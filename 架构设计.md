```kotlin
    override suspend fun doWork(): Result {

            dao.insertAll(entities)
        }
        return Result.success()
    }
}
```

```kotlin
class MlKitOcrService @Inject constructor(
    @ApplicationContext context: Context
) : OcrService {

    private val recognizer = TextRecognition.getClient()

    override suspend fun recognize(imageBitmap: Bitmap): OcrResult = withContext(Dispatchers.IO) {
        val inputImage = InputImage.fromBitmap(imageBitmap, 0)
        val result = recognizer.process(inputImage).await()

        // 解析文本，提取期号和号码
        val text = result.text
        OcrResult(
            period = extractPeriod(text),
            redBalls = extractRedBalls(text),
            blueBall = extractBlueBall(text)
        )
    }

    private fun extractPeriod(text: String): String {
        // 正则匹配 2024XXX
        val regex = Regex("""20\d{2}\d{3}""")
        return regex.find(text)?.value ?: ""
    }

    private fun extractRedBalls(text: String): List<Int> {
        // 提取1-33的数字，最多6个
        val regex = Regex("""\b([1-9]|[12][0-9]|3[0-3])\b""")
        return regex.findAll(text).map { it.value.toInt() }.distinct().take(6)
    }

    private fun extractBlueBall(text: String): Int? {
        // 提取1-16的单个数字
        val regex = Regex("""\b(1[0-6]|[1-9])\b""")
        return regex.find(text)?.value?.toInt()
    }
}
```

```kotlin
// RecognizeLotteryUseCase.kt
class RecognizeLotteryUseCase @Inject constructor(
    private val ocrService: OcrService,
    private val drawRepository: DrawRepository,
    private val recordRepository: LotteryRepository
) {
    suspend operator fun invoke(image: Bitmap): RecognizeResult {
        // 1. OCR 识别
        val ocrResult = ocrService.recognize(image)

        // 2. 查询开奖数据
        val draw = drawRepository.getByPeriod(ocrResult.period)
            ?: throw NotFoundException("未找到期号 ${ocrResult.period} 的开奖数据")

        // 3. 计算中奖
        val matchResult = calculateMatch(ocrResult, draw)

        // 4. 保存记录
        val record = LotteryRecordEntity(
            period = ocrResult.period,
            type = inferType(ocrResult),
            redBalls = ocrResult.redBalls,
            blueBall = ocrResult.blueBall,
            drawRedBalls = draw.redBalls,
            drawBlueBall = draw.blueBall,
            matchedRedCount = matchResult.redCount,
            matchedBlue = matchResult.blueMatch,
            prizeLevel = matchResult.prizeLevel,
            prizeAmount = matchResult.prizeAmount,
            createdAt = System.currentTimeMillis()
        )
        recordRepository.insert(record)

        return RecognizeResult(record = record, match = matchResult)
    }
}
```

```kotlin
@HiltViewModel
class HomeViewModel @Inject constructor(
    private val getStatsUseCase: GetStatsUseCase,
    private val getRecentRecordsUseCase: GetRecentRecordsUseCase
) : ViewModel() {

    private val _state = MutableStateFlow(HomeState())
    val state: StateFlow<HomeState> = _state.asStateFlow()

    init {
        loadData()
    }

    private fun loadData() {
        viewModelScope.launch {
            _state.update { it.copy(isLoading = true) }

            try {
                val stats = getStatsUseCase(Year.now().value)
                val records = getRecentRecordsUseCase(limit = 10)

                _state.update {
                    it.copy(
                        stats = stats,
                        recentRecords = records,
                        isLoading = false
                    )
                }
            } catch (e: Exception) {
                _state.update {
                    it.copy(error = e.message, isLoading = false)
                }
            }
        }
    }
}
```

## 5. 关键算法

### 5.1 中奖计算算法

```kotlin
// PrizeCalculator.kt
object PrizeCalculator {

    // 双色球奖级表 (6+1)
    private val doubleColorRules = listOf(
        PrizeRule(red = 6, blue = 1, level = 1, prize = 5000000), // 一等奖
        PrizeRule(red = 6, blue = 0, level = 2, prize = 200000), // 二等奖
        PrizeRule(red = 5, blue = 1, level = 3, prize = 3000),
        PrizeRule(red = 5, blue = 0, level = 4, prize = 200),
        PrizeRule(red = 4, blue = 1, level = 4, prize = 200),
        PrizeRule(red = 4, blue = 0, level = 5, prize = 10),
        PrizeRule(red = 3, blue = 1, level = 5, prize = 10),
        PrizeRule(red = 2, blue = 1, level = 6, prize = 5),
        PrizeRule(red = 1, blue = 1, level = 6, prize = 5),
        PrizeRule(red = 0, blue = 1, level = 6, prize = 5)
    )

    fun calculate(
        red: List<Int>, blue: Int?,
        drawRed: List<Int>, drawBlue: Int?
    ): PrizeResult {
        val redMatch = red.intersect(drawRed.toSet()).size
        val blueMatch = blue == drawBlue

        // 查找符合条件的最高奖级
        val rule = doubleColorRules.firstOrNull {
            it.red == redMatch && it.blue == (if (it.blue == 1) 1 else 0)
                    && blueMatch == (it.blue == 1)
        }

        return if (rule != null) {
            PrizeResult(level = rule.level, amount = rule.prize.toDouble())
        } else {
            PrizeResult(level = null, amount = 0.0)
        }
    }
}
```

```kotlin
// BackupRepositoryImpl.kt
class BackupRepositoryImpl @Inject constructor(
    private val recordDao: LotteryRecordDao,
    @ApplicationContext private val context: Context
) : BackupRepository {

    override suspend fun exportJson(): File {
        val records = recordDao.getAll().first() // 获取当前所有
        val backup = BackupData(
            version = "1.0",
            exportDate = Instant.now().toString(),
            records = records.map { it.toDomain() }
        )
        val json = GsonBuilder().setPrettyPrinting().create().toJson(backup)
        val file = File(context.getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS),
            "lottery-backup-${LocalDate.now()}.json")
        file.writeText(json)
        return file
    }

    override suspend fun importJson(file: File): Result<Int> {
        val json = file.readText()
        val backup = Gson().fromJson(json, BackupData::class.java)
        val entities = backup.records.map { it.toEntity() }
        recordDao.insertAll(entities)
        return Result.success(entities.size)
    }
}
```

## 9. 性能优化
